{{- /* 用法: {{< card "https://example.com" >}} 或 {{< card url="https://example.com" title="标题" desc="描述" >}} */ -}} {{- $url := .Get 0 | default (.Get "url") -}} {{- $title := .Get "title" | default "" -}} {{- $desc := .Get "desc" | default "" -}} {{- if $url -}} {{- $domain := replaceRE `^https?://([^/]+).*` "$1" $url -}} {{- /* 源站图标：先试 ，再试 /favicon.ico，最后 Google */ -}} {{- $favicon := printf "%s://%s/favicon.ico" (replaceRE `^(https?)://.*` "$1" $url) $domain -}} {{- $gotOGP := false -}} {{- /* 构建时抓取 OGP + */ -}} {{- if not $title -}} {{- $remote := try (resources.GetRemote $url) -}} {{- with $remote -}} {{- with .Value -}} {{- $body := .Content -}} {{- if $body -}} {{- $gotOGP = true -}} {{- /* og:title */ -}} {{- range findRE `]*property="og:title"[^>]*>` $body 1 -}} {{- $title = replaceRE `.*content="([^"]*)".*` "$1" . -}} {{- end -}} {{- /* og:description */ -}} {{- if not $desc -}} {{- range findRE `]*property="og:description"[^>]*>` $body 1 -}} {{- $desc = replaceRE `.*content="([^"]*)".*` "$1" . -}} {{- end -}} {{- end -}} {{- /* fallback: tag */ -}} {{- if not $title -}} {{- range findRE `<title[^>]*>[^<]+` $body 1 -}} {{- $title = replaceRE `]*>([^<]+)` "$1" . -}} {{- end -}} {{- end -}} {{- /* 覆盖默认 /favicon.ico */ -}} {{- $iconMatch := findRE `]*rel=["'][^"']*icon[^"']*["'][^>]*>` $body 1 -}} {{- range $iconMatch -}} {{- $href := findRE `href=["']([^"']+)["']` . 1 -}} {{- range $href -}} {{- $raw := replaceRE `href=["']([^"']+)["']` "$1" . -}} {{- if or (hasPrefix $raw "http://") (hasPrefix $raw "https://") -}} {{- $favicon = $raw -}} {{- else if (hasPrefix $raw "/") -}} {{- $favicon = printf "%s://%s%s" (replaceRE `^(https?)://.*` "$1" $url) $domain $raw -}} {{- else -}} {{- $favicon = printf "%s://%s/%s" (replaceRE `^(https?)://.*` "$1" $url) $domain $raw -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- /* 最终兜底 */ -}} {{- if not $title }}{{ $title = $url }}{{ end -}}
{{ $title }}
{{- if $desc -}}
{{ $desc }}
{{- end -}}
{{ $domain }}
{{- end -}}
